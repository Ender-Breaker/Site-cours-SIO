Cours bonus

Ce cours est hors programme du BTS : il ne compte ni dans ta progression ni au classement. Il est là pour aller plus loin.

Une classe regroupe des données (les attributs) et les fonctions qui agissent dessus (les méthodes). Un objet est un exemplaire de cette classe.

1. D'abord struct : regrouper des données

#include <iostream>
#include <string>
#include <vector>
using namespace std;

struct Produit {
    string nom;
    double prix;
    int stock = 0;                 // valeur par défaut
};

int main() {
    Produit p = {"Souris", 19.9, 4};
    Produit q;                         // nom vide, prix indéterminé, stock 0
    q.nom = "Clavier";
    q.prix = 49;
    vector<Produit> catalogue = {p, q};
    for (const Produit& x : catalogue) {
        cout << x.nom << " : " << x.prix << " € (" << x.stock << ")" << endl;
    }
}

2. Une classe : données privées, méthodes publiques

#include <iostream>
#include <string>
using namespace std;

class Compte {
private:                               // inaccessible de l'extérieur
    string titulaire;
    double solde;

public:                                // l'interface de la classe
    Compte(const string& t, double s) : titulaire(t), solde(s) {}   // constructeur

    void deposer(double montant) {
        if (montant > 0) solde += montant;
    }
    bool retirer(double montant) {
        if (montant > solde) return false;
        solde -= montant;
        return true;
    }
    double getSolde() const { return solde; }      // const : ne modifie pas l'objet
};

int main() {
    Compte c("Léa", 100);
    c.deposer(50);
    cout << c.retirer(500) << " " << c.retirer(30) << endl;
    cout << c.getSolde() << endl;
    // c.solde = 1000000;   // refusé à la compilation : solde est privé
}

C'est l'encapsulation : l'état de l'objet n'est modifiable que par ses méthodes, qui garantissent qu'il reste cohérent (ici, jamais de solde négatif).

La seule différence entre struct et class : dans une struct tout est public par défaut, dans une class tout est privé par défaut.

3. Constructeurs

Le constructeur porte le nom de la classe et n'a pas de type de retour. Il est appelé automatiquement à la création de l'objet. La liste d'initialisation (après les deux-points) donne leur valeur aux attributs.

#include <iostream>
#include <string>
using namespace std;

class Point {
public:
    double x, y;
    Point() : x(0), y(0) {}                          // constructeur par défaut
    Point(double x, double y) : x(x), y(y) {}        // constructeur à deux paramètres
};

int main() {
    Point a;                 // appelle Point()
    Point b(3, 4);           // appelle Point(double, double)
    Point c{5, 6};           // écriture avec accolades, équivalente
    cout << a.x << " " << b.y << " " << c.x << endl;
}

4. Méthodes const, this, membres static

#include <iostream>
#include <string>
using namespace std;

class Joueur {
    string nom;
    int pv;
    static int nombre;                       // partagé par TOUS les joueurs
public:
    Joueur(const string& nom, int pv) : nom(nom), pv(pv) { nombre++; }

    Joueur& blesser(int degats) {
        this->pv -= degats;                  // this : pointeur vers l'objet courant
        if (pv < 0) pv = 0;
        return *this;                        // permet d'enchaîner les appels
    }
    bool estVivant() const { return pv > 0; }      // promet de ne rien modifier
    int getPv() const { return pv; }
    static int combien() { return nombre; }        // appelée sans objet
};
int Joueur::nombre = 0;                      // définition du membre static

int main() {
    Joueur a("Léa", 10), b("Sam", 5);
    a.blesser(3).blesser(2);                     // enchaînement grâce à return *this
    b.blesser(99);
    cout << a.getPv() << " " << b.estVivant() << " " << Joueur::combien() << endl;
}
Marque const tout ce qui ne modifie rien

Un objet reçu par const Joueur& ne peut appeler que des méthodes const. Oublier le const sur un accesseur provoque des erreurs de compilation plus loin, là où on s'y attend le moins.

5. Le destructeur

Le destructeur (~NomDeLaClasse) est appelé automatiquement quand l'objet est détruit : à la fin de son bloc pour une variable locale. Il sert à libérer ce que l'objet a réservé.

#include <iostream>
#include <string>
using namespace std;

class Fichier {
    string nom;
public:
    Fichier(const string& n) : nom(n) { cout << "ouverture de " << nom << endl; }
    ~Fichier() { cout << "fermeture de " << nom << endl; }
};

int main() {
    Fichier a("a.txt");
    {
        Fichier b("b.txt");
        cout << "fin du bloc" << endl;
    }                                  // b est détruit ici
    cout << "fin de main" << endl;
}

6. enum class : une liste de valeurs nommées

#include <iostream>
using namespace std;

enum class Etat { Menu, Jeu, Pause, Fin };

int main() {
    Etat e = Etat::Jeu;
    if (e == Etat::Jeu) cout << "en jeu" << endl;
    switch (e) {
        case Etat::Menu: cout << "menu" << endl; break;
        case Etat::Jeu:  cout << "partie en cours" << endl; break;
        default:         cout << "autre" << endl;
    }
    cout << static_cast<int>(Etat::Pause) << endl;     // 2 : numérotés à partir de 0
}

Des méthodes qui modifient l'état

#include <iostream>
using namespace std;

class Chrono {
    int secondes;
public:
    Chrono() : secondes(0) {}
    void avancer(int s) { if (s > 0) secondes += s; }
    void remettreAZero() { secondes = 0; }
    int total() const { return secondes; }
};

int main() {
    Chrono c;
    c.avancer(75);
    c.avancer(-10);        // ignoré : la classe se protège
    cout << c.total() << endl;
}

Une méthode d'affichage

const partout où l'objet n'est pas modifié. Ajoute une méthode heures().

#include <iostream>
#include <string>
using namespace std;

class Chrono {
    int secondes;
public:
    Chrono() : secondes(0) {}
    void avancer(int s) { if (s > 0) secondes += s; }
    int minutes() const { return secondes / 60; }
    string texte() const {
        return to_string(minutes()) + " min " + to_string(secondes % 60) + " s";
    }
};

int main() {
    Chrono c;
    c.avancer(75);
    c.avancer(3600);
    cout << c.texte() << endl;
}
