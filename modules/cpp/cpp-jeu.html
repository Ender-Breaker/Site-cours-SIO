Cours bonus

Ce cours est hors programme du BTS : il ne compte ni dans ta progression ni au classement. Il est là pour aller plus loin.

La plupart des grands jeux sont écrits en C++ (Unreal Engine, les moteurs maison des studios) : il est rapide et contrôle précisément la mémoire. Dans le navigateur, le C++ est compilé en WebAssembly, un format binaire exécuté presque aussi vite que du code natif. C'est ce que fait le labo.

Où tester ?

Les jeux ont besoin d'un écran : ils se lancent dans le bac à sable du labo. Copie les exemples de ce chapitre et modifie-les.

1. La boucle de jeu

Le programme n'a pas de main : c'est le moteur qui appelle tes fonctions.

#include "jeu.h"                 // l'API du labo

double x = 200;                  // l'état du jeu

void setup() {                   // une fois, au démarrage
    x = 50;
}

void update(double dt) {         // 60 fois par seconde : la logique
    if (touche("droite")) x += 200 * dt;      // 200 pixels par seconde
}

void draw() {                    // 60 fois par seconde : l'affichage
    effacer("black");
    rect(x, 140, 40, 40, "gold");
}

dt est le temps écoulé depuis l'image précédente (environ 0,016 s). Multiplier une vitesse par dt donne un mouvement identique sur toutes les machines.

2. Des entités avec des struct

#include "jeu.h"
#include <vector>

struct Balle {
    double x, y, vx, vy;
    void avancer() {
        x += vx; y += vy;
        if (x < 10 || x > LARGEUR - 10) vx = -vx;
        if (y < 10 || y > HAUTEUR - 10) vy = -vy;
    }
};

std::vector<Balle> balles;

void setup() {
    for (int i = 0; i < 5; i++)
        balles.push_back({ (double)hasard(20, 460), (double)hasard(20, 300), 3.0, 2.0 });
}

void update(double dt) {
    for (Balle& b : balles) b.avancer();          // Balle& : on modifie l'original
}

void draw() {
    effacer("#16182b");
    for (const Balle& b : balles) cercle(b.x, b.y, 10, "gold");
}

3. Gravité et saut

double y = 260, vy = 0;
const double SOL = 260;

void update(double dt) {
    if (toucheAppuyee("espace") && y >= SOL) vy = -9;     // saut seulement au sol
    vy += 0.5;                                            // gravité
    y += vy;
    if (y > SOL) { y = SOL; vy = 0; }
}

4. Collisions

Le test de chevauchement de deux rectangles (AABB), exécutable ici car il n'a pas besoin d'écran :

#include <iostream>
using namespace std;

struct Boite { double x, y, l, h; };

bool collision(const Boite& a, const Boite& b) {
    return a.x < b.x + b.l && b.x < a.x + a.l
        && a.y < b.y + b.h && b.y < a.y + a.h;
}

int main() {
    Boite joueur{100, 100, 40, 40};
    cout << collision(joueur, {130, 120, 20, 20}) << endl;     // 1 : se chevauchent
    cout << collision(joueur, {300, 100, 20, 20}) << endl;     // 0 : trop loin
}

5. Créer et supprimer des objets

On supprime avec l'idiome *erase-remove* : remove_if regroupe les éléments à garder, erase coupe le reste.

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Tir { double x, y; };

int main() {
    vector<Tir> tirs = {{10, 50}, {20, -5}, {30, 120}, {40, -30}};
    for (Tir& t : tirs) t.y -= 10;
    tirs.erase(remove_if(tirs.begin(), tirs.end(), [](const Tir& t) { return t.y < 0; }), tirs.end());
    cout << tirs.size() << " tirs encore à l'écran" << endl;
}

6. Les états du jeu

enum class Etat { Menu, Jeu, Fin };
Etat etat = Etat::Menu;
int score = 0;

void update(double dt) {
    switch (etat) {
        case Etat::Menu: if (toucheAppuyee("espace")) { etat = Etat::Jeu; score = 0; } break;
        case Etat::Jeu:  score++; if (score > 300) etat = Etat::Fin; break;
        case Etat::Fin:  if (toucheAppuyee("espace")) etat = Etat::Menu; break;
    }
}

void draw() {
    effacer("black");
    if (etat == Etat::Menu) texte("ESPACE pour jouer", 130, 150, 24, "white");
    else if (etat == Etat::Jeu) texte("Score : " + std::to_string(score), 16, 16, 24, "white");
    else texte("Perdu ! Score : " + std::to_string(score), 130, 150, 24, "tomato");
}

Le vaisseau

Colle ce code dans le bac à sable.

#include "jeu.h"
#include <vector>
#include <algorithm>
#include <string>

struct Boite { double x, y, l, h; };
Boite vaisseau{220, 280, 40, 20};

void update(double dt) {
    if (touche("gauche")) vaisseau.x -= 5;
    if (touche("droite")) vaisseau.x += 5;
    vaisseau.x = std::clamp(vaisseau.x, 0.0, LARGEUR - vaisseau.l);     // reste dans l'écran
}

void draw() {
    effacer("#0d0e1a");
    rect(vaisseau.x, vaisseau.y, vaisseau.l, vaisseau.h, "deepskyblue");
}

La collision et la fin de partie

bool perdu = false;
int score = 0;

    return a.x < b.x + b.l && b.x < a.x + a.l && a.y < b.y + b.h && b.y < a.y + a.h;
}

// au début de update :
    if (perdu) return;
    score++;
// après le déplacement des météorites :
    for (const Boite& m : meteores) if (collision(vaisseau, m)) perdu = true;
// à la fin de draw :
    texte("Score : " + std::to_string(score), 12, 12, 20, "white");
    if (perdu) texte("PERDU", 190, 140, 36, "tomato");
