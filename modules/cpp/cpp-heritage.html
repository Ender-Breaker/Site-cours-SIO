1. Hériter d'une classe

Une classe fille (ou dérivée) reprend les attributs et les méthodes de sa classe mère (ou de base), et peut en ajouter.

#include <iostream>
#include <string>
using namespace std;

class Personnage {
protected:                          // accessible aux classes filles
    string nom;
    int pv;
public:
    Personnage(const string& nom, int pv) : nom(nom), pv(pv) {}
    void blesser(int d) { pv -= d; if (pv < 0) pv = 0; }
    void afficher() const { cout << nom << " : " << pv << " PV" << endl; }
};

class Guerrier : public Personnage {              // Guerrier EST UN Personnage
    int force;
public:
    Guerrier(const string& nom, int force)
        : Personnage(nom, 20), force(force) {}    // appelle d'abord le constructeur de la mère
    void frapper(Personnage& cible) { cible.blesser(force); }
};

int main() {
    Guerrier g("Conan", 7);
    Personnage p("Gobelin", 10);
    g.frapper(p);
    g.afficher();            // méthode héritée
    p.afficher();
}

L'héritage exprime une relation « est un » : un Guerrier est un Personnage. Si la relation est « a un » (une voiture a un moteur), on utilise un attribut, pas l'héritage.

2. Redéfinir une méthode : virtual et override

#include <iostream>
#include <string>
using namespace std;

class Animal {
protected:
    string nom;
public:
    Animal(const string& nom) : nom(nom) {}
    virtual string crier() const { return "..."; }       // virtual : redéfinissable
    void presenter() const { cout << nom << " fait " << crier() << endl; }
    virtual ~Animal() = default;                         // destructeur virtuel : voir plus bas
};

class Chien : public Animal {
public:
    Chien(const string& nom) : Animal(nom) {}
    string crier() const override { return "Ouaf"; }     // override : vérifié par le compilateur
};

class Chat : public Animal {
public:
    using Animal::Animal;                                // réutilise les constructeurs de la mère
    string crier() const override { return "Miaou"; }
};

int main() {
    Chien rex("Rex");
    Chat tigrou("Tigrou");
    rex.presenter();
    tigrou.presenter();
}

presenter est écrite une seule fois dans Animal, mais appelle le crier de la vraie classe de l'objet : c'est le polymorphisme.

Sans virtual, pas de polymorphisme

Si crier n'est pas virtual dans la classe mère, c'est toujours la version de la mère qui est appelée à travers une référence ou un pointeur de type Animal. Et sans override, une faute de frappe dans la signature crée silencieusement une nouvelle méthode au lieu de redéfinir l'ancienne.

3. Le polymorphisme passe par les pointeurs et les références

#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Forme {
public:
    virtual double aire() const = 0;       // = 0 : méthode virtuelle PURE
    virtual string nom() const = 0;
    virtual ~Forme() = default;
};

class Rectangle : public Forme {
    double l, h;
public:
    Rectangle(double l, double h) : l(l), h(h) {}
    double aire() const override { return l * h; }
    string nom() const override { return "rectangle"; }
};

class Cercle : public Forme {
    double r;
public:
    Cercle(double r) : r(r) {}
    double aire() const override { return 3.14159 * r * r; }
    string nom() const override { return "cercle"; }
};

void decrire(const Forme& f) {             // accepte n'importe quelle Forme
    cout << f.nom() << " d'aire " << f.aire() << endl;
}

int main() {
    Rectangle r(3, 4);
    Cercle c(1);
    decrire(r);
    decrire(c);

    vector<Forme*> formes = {&r, &c};          // une collection de formes différentes
    double total = 0;
    for (const Forme* f : formes) total += f->aire();
    cout << "Aire totale : " << total << endl;
}

Une classe qui possède au moins une méthode virtuelle pure est abstraite : on ne peut pas créer d'objet Forme, seulement des objets de ses classes filles. C'est l'équivalent d'une interface.

Le découpage (*slicing*)

vector<Forme> ne peut pas contenir des rectangles : copier un Rectangle dans une variable de type Forme ne garde que la partie mère. Pour le polymorphisme, il faut toujours des pointeurs ou des références.

4. Appeler la version de la mère

#include <iostream>
#include <string>
using namespace std;

class Employe {
protected:
    string nom;
    double salaire;
public:
    Employe(const string& n, double s) : nom(n), salaire(s) {}
    virtual double paie() const { return salaire; }
    virtual ~Employe() = default;
};

class Commercial : public Employe {
    double ventes;
public:
    Commercial(const string& n, double s, double v) : Employe(n, s), ventes(v) {}
    double paie() const override {
        return Employe::paie() + ventes * 0.05;     // la paie de base + une commission
    }
};

int main() {
    Commercial c("Sam", 2000, 10000);
    const Employe& e = c;
    cout << e.paie() << endl;
}

5. Pourquoi un destructeur virtuel ?

Quand on détruit un objet de classe fille à travers un pointeur de la classe mère, seul un destructeur virtual garantit que le destructeur de la fille est appelé. Règle simple : une classe avec au moins une méthode virtuelle a un destructeur virtuel.

#include <iostream>
using namespace std;

class Base {
public:
    virtual ~Base() { cout << "destruction de Base" << endl; }
};
class Derivee : public Base {
public:
    ~Derivee() override { cout << "destruction de Derivee" << endl; }
};

int main() {
    Base* p = new Derivee();
    delete p;              // appelle ~Derivee puis ~Base grâce à virtual
}

Deux classes concrètes

#include <iostream>
#include <string>
using namespace std;

class Notification {
protected:
    string message;
public:
    Notification(const string& m) : message(m) {}
    virtual void envoyer() const = 0;
    virtual ~Notification() = default;
};
class Email : public Notification {
    string adresse;
public:
    Email(const string& m, const string& a) : Notification(m), adresse(a) {}
    void envoyer() const override { cout << "Mail à " << adresse << " : " << message << endl; }
};
class Sms : public Notification {
    string numero;
public:
    Sms(const string& m, const string& n) : Notification(m), numero(n) {}
    void envoyer() const override { cout << "SMS au " << numero << " : " << message << endl; }
};

int main() {
    Email e("Serveur en panne", "admin@exemple.fr");
    Sms s("Serveur en panne", "0601020304");
    e.envoyer();
    s.envoyer();
}

Les traiter toutes de la même façon

Le code qui envoie ne connaît que Notification. Ajouter un nouveau type (une notification Slack) ne demande aucune modification de cette boucle : essaie.

#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Notification {
protected:
    string message;
public:
    Notification(const string& m) : message(m) {}
    virtual void envoyer() const = 0;
    virtual ~Notification() = default;
};
class Email : public Notification {
public:
    using Notification::Notification;
    void envoyer() const override { cout << "[mail] " << message << endl; }
};
class Sms : public Notification {
public:
    using Notification::Notification;
    void envoyer() const override { cout << "[sms] " << message << endl; }
};

int main() {
    Email e("Sauvegarde terminée");
    Sms s("Disque plein");
    vector<const Notification*> file = {&e, &s, &e};
    for (const Notification* n : file) n->envoyer();
}

À retenir

class Fille : public Mere { … }; ; le constructeur de la fille appelle celui de la mère dans sa liste d'initialisation.

virtual dans la mère, override dans la fille.

virtual … = 0; : méthode pure, classe abstraite (non instanciable).

Le polymorphisme exige un pointeur ou une référence vers la classe mère.

Destructeur virtual dès qu'il y a une méthode virtuelle.
