1. Le problème : le même code pour plusieurs types

Écrire maxi pour les int, puis pour les double, puis pour les string, c'est trois fois le même code. Un template est un modèle de fonction dont le type est un paramètre.

#include <iostream>
#include <string>
using namespace std;

template <typename T>
T maxi(T a, T b) {
    return a > b ? a : b;
}

int main() {
    cout << maxi(3, 8) << endl;                          // T = int, déduit
    cout << maxi(2.5, 1.5) << endl;                      // T = double
    cout << maxi(string("abc"), string("abd")) << endl;  // T = string
    cout << maxi<double>(3, 2.5) << endl;                // T imposé : sinon ambigu (int et double)
}

Le compilateur fabrique une version de la fonction pour chaque type utilisé. Il n'y a aucun coût à l'exécution.

2. Templates et conteneurs

#include <iostream>
#include <vector>
#include <string>
using namespace std;

template <typename T>
void afficher(const vector<T>& v) {
    for (const T& x : v) cout << x << " ";
    cout << endl;
}

template <typename T>
T somme(const vector<T>& v) {
    T total = T();                 // valeur par défaut du type : 0, 0.0, ""
    for (const T& x : v) total += x;
    return total;
}

int main() {
    afficher(vector<int>{1, 2, 3});
    afficher(vector<string>{"a", "b"});
    cout << somme(vector<double>{1.5, 2.5}) << " " << somme(vector<string>{"ab", "cd"}) << endl;
}

3. Classes template

vector<int>, map<string, int>, unique_ptr<Forme> sont des classes template. On peut écrire les siennes :

#include <iostream>
#include <vector>
#include <string>
using namespace std;

template <typename T>
class Pile {
    vector<T> elements;
public:
    void empiler(const T& x) { elements.push_back(x); }
    T depiler() {
        T haut = elements.back();
        elements.pop_back();
        return haut;
    }
    bool vide() const { return elements.empty(); }
    size_t taille() const { return elements.size(); }
};

int main() {
    Pile<int> p;
    p.empiler(1); p.empiler(2); p.empiler(3);
    cout << p.depiler() << " " << p.depiler() << " " << p.taille() << endl;

    Pile<string> mots;
    mots.empiler("monde"); mots.empiler("bonjour");
    while (!mots.vide()) cout << mots.depiler() << " ";
    cout << endl;
}
Les messages d'erreur des templates

Une erreur dans l'utilisation d'un template produit souvent des dizaines de lignes. Remonte toujours à la première erreur : elle dit quel type a été utilisé et quelle opération lui manque.

4. Surcharger les opérateurs

On peut donner un sens à +, ==, <, <<… pour ses propres classes. Un opérateur n'est qu'une fonction au nom spécial.

#include <iostream>
using namespace std;

struct Vec {
    double x, y;

    Vec operator+(const Vec& o) const { return {x + o.x, y + o.y}; }
    Vec operator*(double k) const { return {x * k, y * k}; }
    bool operator==(const Vec& o) const { return x == o.x && y == o.y; }
    Vec& operator+=(const Vec& o) { x += o.x; y += o.y; return *this; }
};

ostream& operator<<(ostream& flux, const Vec& v) {      // pour cout << v
    return flux << "(" << v.x << ", " << v.y << ")";
}

int main() {
    Vec a{1, 2}, b{3, 4};
    cout << a + b << endl;
    cout << a * 10 << endl;
    cout << (a == Vec{1, 2}) << endl;
    a += b;
    cout << a << endl;
}

operator<< est une fonction hors de la classe : son opérande de gauche est le flux (cout), pas l'objet. Elle renvoie le flux pour permettre d'enchaîner cout << a << b.

5. Rendre une classe triable

sort, set et map ont besoin de savoir comparer : il suffit de définir operator<.

#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <set>
using namespace std;

struct Version {
    int majeur, mineur;
    bool operator<(const Version& o) const {
        if (majeur != o.majeur) return majeur < o.majeur;
        return mineur < o.mineur;
    }
};
ostream& operator<<(ostream& f, const Version& v) { return f << v.majeur << "." << v.mineur; }

int main() {
    vector<Version> v = {{2, 1}, {1, 9}, {2, 0}, {1, 10}};
    sort(v.begin(), v.end());                 // utilise operator<
    for (const Version& x : v) cout << x << " ";
    cout << endl;
    set<Version> uniques(v.begin(), v.end());
    cout << uniques.size() << " " << *uniques.rbegin() << endl;     // la plus grande
}

#include <iostream>

struct Fraction {

Simplifier avec le PGCD

Le constructeur garantit qu'une fraction est toujours simplifiée : c'est un *invariant* de la classe. Ajoute la multiplication.

#include <iostream>
#include <numeric>
using namespace std;

struct Fraction {
    int num, den;
    Fraction(int n, int d) : num(n / gcd(n, d)), den(d / gcd(n, d)) {}      // gcd : <numeric>
    Fraction operator+(const Fraction& o) const {
        return Fraction(num * o.den + o.num * den, den * o.den);
    }
    bool operator==(const Fraction& o) const { return num == o.num && den == o.den; }
};
ostream& operator<<(ostream& f, const Fraction& q) { return f << q.num << "/" << q.den; }

int main() {
    cout << Fraction(1, 2) + Fraction(1, 6) << endl;             // 2/3
    cout << (Fraction(2, 4) == Fraction(1, 2)) << endl;          // 1
}
