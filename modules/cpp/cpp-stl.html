La STL (*Standard Template Library*) est la boîte à outils du C++ : des conteneurs pour ranger les données, des algorithmes pour les traiter, et des itérateurs pour relier les deux. La connaître évite de réécrire (mal) ce qui existe déjà.

1. map : le dictionnaire

map<Clé, Valeur> associe une valeur à chaque clé, et garde les clés triées.

#include <iostream>
#include <map>
#include <string>
using namespace std;

int main() {
    map<string, int> stock;
    stock["souris"] = 3;
    stock["clavier"] = 2;
    stock["souris"] += 5;                       // modifie
    stock["ecran"]++;                           // clé absente : créée avec la valeur 0, puis +1
    cout << stock["souris"] << " " << stock.size() << endl;

    for (const auto& [nom, quantite] : stock) {     // parcours dans l'ordre des clés
        cout << nom << " : " << quantite << endl;
    }
    if (stock.count("webcam") == 0) cout << "pas de webcam" << endl;
    if (stock.contains("clavier")) cout << "clavier présent" << endl;     // C++20
    stock.erase("clavier");
    cout << stock.size() << endl;
}
m[cle] crée la clé

Lire stock["webcam"] ajoute la clé avec la valeur 0 si elle n'existe pas. Pour tester la présence sans rien créer, utilise count, contains ou find.

2. set, unordered_map, pair

#include <iostream>
#include <set>
#include <unordered_map>
#include <string>
#include <utility>
using namespace std;

int main() {
    set<int> vus = {5, 2, 5, 8, 2};             // ensemble : pas de doublon, trié
    vus.insert(1);
    for (int v : vus) cout << v << " ";
    cout << "(" << vus.size() << " valeurs)" << endl;

    unordered_map<string, double> prix;         // table de hachage : accès très rapide, pas d'ordre
    prix["souris"] = 19.9;
    cout << prix["souris"] << endl;

    pair<string, int> p = {"Léa", 17};          // deux valeurs réunies
    cout << p.first << " " << p.second << endl;
    auto [nom, age] = p;                        // décomposition (C++17)
    cout << nom << " a " << age << " ans" << endl;
}

3. Les lambdas : des fonctions écrites sur place

#include <iostream>
using namespace std;

int main() {
    auto carre = [](int x) { return x * x; };          // [] capture, () paramètres, {} corps
    cout << carre(7) << endl;

    int seuil = 10;
    auto auDessus = [seuil](int x) { return x > seuil; };   // capture seuil par copie
    cout << auDessus(15) << " " << auDessus(3) << endl;

    int total = 0;
    auto ajouter = [&total](int x) { total += x; };         // capture par référence : peut modifier
    ajouter(5);
    ajouter(7);
    cout << total << endl;
}
CaptureSignifie
[]rien
[x]une copie de x
[&x]x par référence
[=]tout ce qui est utilisé, par copie
[&]tout ce qui est utilisé, par référence

4. Les algorithmes

Ils travaillent sur un intervalle [begin, end) et prennent souvent une lambda.

#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

int main() {
    vector<int> v = {12, 5, 19, 8, 10};

    sort(v.begin(), v.end());                                         // croissant
    sort(v.begin(), v.end(), [](int a, int b) { return a > b; });     // décroissant
    for (int x : v) cout << x << " ";
    cout << endl;

    cout << count_if(v.begin(), v.end(), [](int x) { return x >= 10; }) << " notes >= 10" << endl;
    cout << accumulate(v.begin(), v.end(), 0) << endl;                // somme (<numeric>)
    cout << *max_element(v.begin(), v.end()) << " " << *min_element(v.begin(), v.end()) << endl;

    auto it = find(v.begin(), v.end(), 19);                           // renvoie un itérateur
    if (it != v.end()) cout << "19 trouvé à l'indice " << it - v.begin() << endl;
    cout << any_of(v.begin(), v.end(), [](int x) { return x < 6; }) << endl;
    cout << all_of(v.begin(), v.end(), [](int x) { return x > 0; }) << endl;
}

Transformer et filtrer

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    vector<int> v = {1, 2, 3, 4, 5, 6};

    vector<int> doubles(v.size());
    transform(v.begin(), v.end(), doubles.begin(), [](int x) { return x * 2; });

    vector<int> pairs;
    copy_if(v.begin(), v.end(), back_inserter(pairs), [](int x) { return x % 2 == 0; });

    v.erase(remove_if(v.begin(), v.end(), [](int x) { return x > 4; }), v.end());   // supprime selon un critère

    for (int x : doubles) cout << x << " ";
    cout << "| ";
    for (int x : pairs) cout << x << " ";
    cout << "| ";
    for (int x : v) cout << x << " ";
    cout << endl;
}

5. Trier des objets

#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

struct Eleve {
    string nom;
    double note;
};

int main() {
    vector<Eleve> classe = {{"Léa", 15}, {"Sam", 9}, {"Zoé", 18}};
    sort(classe.begin(), classe.end(), [](const Eleve& a, const Eleve& b) {
        return a.note > b.note;                  // vrai si a doit passer avant b
    });
    for (const Eleve& e : classe) cout << e.nom << " " << e.note << endl;
}

Compter avec une map

compte[mot]++ suffit : une clé absente démarre à 0.

#include <iostream>
#include <string>
#include <map>
using namespace std;

int main() {
    map<string, int> compte;
    string mot;
    while (cin >> mot) compte[mot]++;
    for (const auto& [m, n] : compte) cout << m << " : " << n << endl;
}

Classer par fréquence

Une map est triée par clé. Pour trier par valeur, on copie les paires dans un vector.

#include <iostream>
#include <string>
#include <map>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    map<string, int> compte;
    string mot;
    while (cin >> mot) compte[mot]++;
    vector<pair<string, int>> liste(compte.begin(), compte.end());
    sort(liste.begin(), liste.end(), [](const auto& a, const auto& b) { return a.second > b.second; });
    for (const auto& [m, n] : liste) cout << n << " x " << m << endl;
}
