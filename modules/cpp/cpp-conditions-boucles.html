1. if, else if, else

#include <iostream>
using namespace std;

int main() {
    double note;
    cin >> note;
    if (note >= 16) {
        cout << "Très bien" << endl;
    } else if (note >= 10) {
        cout << "Admis" << endl;
    } else {
        cout << "Recalé" << endl;
    }
}

La condition est entre parenthèses, le bloc entre accolades. Les accolades sont facultatives pour une seule instruction, mais mets-les toujours : cela évite des bugs célèbres.

int main() {
    int age = 17;

= n'est pas ==

if (x = 5) affecte 5 à x puis teste 5, qui est toujours vrai. Le compilateur ne te prévient que si tu actives les avertissements (-Wall). En C++, tout nombre différent de 0 est vrai.

3. L'opérateur ternaire et switch

#include <iostream>
#include <string>
using namespace std;

int main() {
    int jour;
    cin >> jour;
    string type = (jour >= 6) ? "week-end" : "semaine";   // condition ? si vrai : si faux
    cout << type << endl;

    switch (jour) {
        case 1: cout << "lundi" << endl; break;
        case 2: cout << "mardi" << endl; break;
        case 3: cout << "mercredi" << endl; break;
        case 6:
        case 7: cout << "repos" << endl; break;      // deux cas, même traitement
        default: cout << "autre jour" << endl;
    }
}

switch compare une valeur entière (ou un char, un enum) à des constantes. Sans break, l'exécution continue dans le cas suivant.

4. while et do … while

#include <iostream>
using namespace std;

int main() {
    int n = 1;
    while (n < 100) {        // teste AVANT chaque tour
        n *= 2;
    }
    cout << n << endl;

    int essais = 0;
    do {                     // teste APRÈS : au moins un tour
        essais++;
    } while (essais * essais <= 50);
    cout << essais << endl;
}

5. for

for (départ; condition; évolution) : le départ s'exécute une fois, la condition avant chaque tour, l'évolution après chaque tour.

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {
        cout << i << " ";
    }
    cout << endl;
    for (int i = 10; i > 0; i -= 3) cout << i << " ";
    cout << endl;
    for (int ligne = 1; ligne <= 3; ligne++) {          // boucles imbriquées
        for (int col = 1; col <= ligne; col++) cout << "*";
        cout << endl;
    }
}

6. break et continue

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 10; i++) {
        if (i % 2 == 0) continue;    // passe directement au tour suivant
        if (i > 7) break;            // sort de la boucle
        cout << i << " ";
    }
    cout << endl;
}
Les boucles infinies

Une boucle dont la condition ne devient jamais fausse ne s'arrête pas : vérifie toujours que la variable testée change dans la boucle. Dans le labo, le programme est coupé au bout de quelques secondes.

TutorielLe jeu du « plus ou moins »

Une seule comparaison

#include <iostream>
using namespace std;

int main() {
    const int SECRET = 42;
    int essai;
    cin >> essai;
    if (essai < SECRET) cout << "Trop petit" << endl;
    else if (essai > SECRET) cout << "Trop grand" << endl;
    else cout << "Gagné !" << endl;
}

Répéter jusqu'à trouver

On ne sait pas combien d'essais il faudra, mais il y en a au moins un : do … while.

#include <iostream>
using namespace std;

int main() {
    const int SECRET = 42;
    int essai;
    do {
        cin >> essai;
        if (essai < SECRET) cout << "Trop petit" << endl;
        else if (essai > SECRET) cout << "Trop grand" << endl;
    } while (essai != SECRET);
    cout << "Gagné !" << endl;
}

Compter les essais

#include <iostream>
using namespace std;

int main() {
    const int SECRET = 42;
    int essai, essais = 0;
    do {
        cin >> essai;
        essais++;
        if (essai < SECRET) cout << "Trop petit" << endl;
        else if (essai > SECRET) cout << "Trop grand" << endl;
    } while (essai != SECRET);
    cout << "Gagné ! Essais : " << essais << endl;
}
