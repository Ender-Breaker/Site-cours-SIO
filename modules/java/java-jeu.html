1. La boucle de jeu

Ta classe hérite de Jeu et redéfinit trois méthodes. C'est l'héritage et le polymorphisme en action : le moteur ne connaît que Jeu, et appelle pourtant tes méthodes.

public class Main extends Jeu {
    double x = 200;                          // l'état du jeu : des attributs

    public void setup() {                    // une fois, au démarrage
        x = 50;
    }

    public void update(double dt) {          // 60 fois par seconde : la logique
        if (touche("droite")) x += 200 * dt; // 200 pixels par seconde
    }

    public void draw() {                     // 60 fois par seconde : l'affichage
        effacer("black");
        rect(x, 140, 40, 40, "gold");
    }

    public static void main(String[] args) {
        lancer(new Main());                  // démarre le moteur
    }
}

2. Des entités : une classe par type d'objet

import java.util.ArrayList;
import java.util.List;

class Balle {
    double x, y, vx, vy;
    Balle(double x, double y, double vx, double vy) { this.x = x; this.y = y; this.vx = vx; this.vy = vy; }

    void avancer() {
        x += vx; y += vy;
        if (x < 10 || x > Jeu.LARGEUR - 10) vx = -vx;
        if (y < 10 || y > Jeu.HAUTEUR - 10) vy = -vy;
    }
}

public class Main extends Jeu {
    List<Balle> balles = new ArrayList<>();

    public void setup() {
        for (int i = 0; i < 5; i++) balles.add(new Balle(hasard(20, 460), hasard(20, 300), 3, 2));
    }
    public void update(double dt) {
        for (Balle b : balles) b.avancer();
    }
    public void draw() {
        effacer("#16182b");
        for (Balle b : balles) cercle(b.x, b.y, 10, "gold");
    }
    public static void main(String[] args) { lancer(new Main()); }
}

double y = 260, vy = 0;
final double SOL = 260;

public void update(double dt) {
    if (toucheAppuyee("espace") && y >= SOL) vy = -9;     // saut seulement au sol
    vy += 0.5;                                            // gravité
    y += vy;
    if (y > SOL) { y = SOL; vy = 0; }
}

4. Collisions

Le test de chevauchement de deux rectangles, exécutable ici car il n'a pas besoin d'écran :

record Boite(double x, double y, double l, double h) {
    boolean touche(Boite o) {
        return x < o.x + o.l && o.x < x + l && y < o.y + o.h && o.y < y + h;
    }
}

public class Main {
    public static void main(String[] args) {
        Boite joueur = new Boite(100, 100, 40, 40);
        System.out.println(joueur.touche(new Boite(130, 120, 20, 20)));     // true : se chevauchent
        System.out.println(joueur.touche(new Boite(300, 100, 20, 20)));     // false : trop loin
    }
}

5. Créer et supprimer des objets

import java.util.*;

public class Main {
    public static void main(String[] args) {
        List<double[]> tirs = new ArrayList<>(List.of(new double[]{10, 50}, new double[]{20, -5}, new double[]{30, 120}));
        for (double[] t : tirs) t[1] -= 10;            // chaque tir monte
        tirs.removeIf(t -> t[1] < 0);                  // retire ceux qui sont sortis
        System.out.println(tirs.size() + " tirs encore à l'écran");
    }
}

6. Les états du jeu avec un enum

enum Etat { MENU, JEU, FIN }

public class Main extends Jeu {
    Etat etat = Etat.MENU;
    int score = 0;

    public void update(double dt) {
        switch (etat) {
            case MENU -> { if (toucheAppuyee("espace")) { etat = Etat.JEU; score = 0; } }
            case JEU  -> { score++; if (score > 300) etat = Etat.FIN; }
            case FIN  -> { if (toucheAppuyee("espace")) etat = Etat.MENU; }
        }
    }

    public void draw() {
        effacer("black");
        switch (etat) {
            case MENU -> texte("ESPACE pour jouer", 130, 150, 24, "white");
            case JEU  -> texte("Score : " + score, 16, 16, 24, "white");
            case FIN  -> texte("Perdu ! Score : " + score, 130, 150, 24, "tomato");
        }
    }
    public static void main(String[] args) { lancer(new Main()); }
}

import java.util.ArrayList;
import java.util.List;

public class Main extends Jeu {
    double vx = 220;                              // position du vaisseau

    public void update(double dt) {
        if (touche("gauche")) vx -= 5;
        if (touche("droite")) vx += 5;
        vx = Math.max(0, Math.min(LARGEUR - 40, vx));      // reste dans l'écran
    }

    public void draw() {
        effacer("#0d0e1a");
        rect(vx, 280, 40, 20, "deepskyblue");
    }

    public static void main(String[] args) { lancer(new Main()); }
}
