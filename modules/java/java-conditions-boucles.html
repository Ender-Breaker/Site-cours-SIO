1. if, else if, else

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double note = sc.nextDouble();
        if (note >= 16) {
            System.out.println("Très bien");
        } else if (note >= 10) {
            System.out.println("Admis");
        } else {
            System.out.println("Recalé");
        }
    }
}

La condition est entre parenthèses et doit être un boolean : contrairement au C ou au JavaScript, if (x) avec un entier ne compile pas.

Comparer des String : equals, jamais ==

== entre deux objets teste s'il s'agit du même objet en mémoire, pas s'ils ont le même contenu. Pour du texte, écris toujours a.equals(b). C'est le bug numéro 1 des débutants en Java.

3. Le ternaire et switch

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int jour = sc.nextInt();
        String type = (jour >= 6) ? "week-end" : "semaine";     // condition ? si vrai : si faux
        System.out.println(type);

        switch (jour) {                       // forme classique : un break par cas
            case 1: System.out.println("lundi"); break;
            case 2: System.out.println("mardi"); break;
            case 6:
            case 7: System.out.println("repos"); break;
            default: System.out.println("autre jour");
        }

        String nom = switch (jour) {          // forme moderne (Java 14+) : pas de break, renvoie une valeur
            case 1 -> "lundi";
            case 2 -> "mardi";
            case 3 -> "mercredi";
            case 6, 7 -> "repos";
            default -> "autre jour";
        };
        System.out.println(nom);
    }
}

switch fonctionne avec les entiers, les char, les String et les enum.

4. while et do … while

public class Main {
    public static void main(String[] args) {
        int n = 1;
        while (n < 100) {        // teste AVANT chaque tour
            n *= 2;
        }
        System.out.println(n);

        int essais = 0;
        do {                     // teste APRÈS : au moins un tour
            essais++;
        } while (essais * essais <= 50);
        System.out.println(essais);
    }
}

5. for et for-each

public class Main {
    public static void main(String[] args) {
        for (int i = 1; i <= 5; i++) {
            System.out.print(i + " ");
        }
        System.out.println();
        for (int i = 10; i > 0; i -= 3) System.out.print(i + " ");
        System.out.println();

        int[] notes = {12, 15, 9};
        for (int note : notes) {             // for-each : « pour chaque note dans notes »
            System.out.print(note + " ");
        }
        System.out.println();
        for (char c : "Java".toCharArray()) System.out.print(c + "-");
        System.out.println();
    }
}

6. break et continue

public class Main {
    public static void main(String[] args) {
        for (int i = 1; i <= 10; i++) {
            if (i % 2 == 0) continue;    // passe au tour suivant
            if (i > 7) break;            // sort de la boucle
            System.out.print(i + " ");
        }
        System.out.println();
    }
}

        final int SECRET = 42;
        Scanner sc = new Scanner(System.in);
        int essai = sc.nextInt();
        if (essai < SECRET) System.out.println("Trop petit");

Répéter jusqu'à trouver

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        final int SECRET = 42;
        Scanner sc = new Scanner(System.in);
        int essai;
        do {
            essai = sc.nextInt();
            if (essai < SECRET) System.out.println("Trop petit");
            else if (essai > SECRET) System.out.println("Trop grand");
        } while (essai != SECRET);
        System.out.println("Gagné !");
    }
}

Compter les essais

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        final int SECRET = 42;
        Scanner sc = new Scanner(System.in);
        int essai, essais = 0;
        do {
            essai = sc.nextInt();
            essais++;
            if (essai < SECRET) System.out.println("Trop petit");
            else if (essai > SECRET) System.out.println("Trop grand");
        } while (essai != SECRET);
        System.out.println("Gagné ! Essais : " + essais);
    }
}
