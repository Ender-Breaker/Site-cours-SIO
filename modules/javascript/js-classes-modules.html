1. Les classes

class Compte {
  #solde = 0;                            // # : attribut privé
  constructor(titulaire, solde = 0) {
    this.titulaire = titulaire;
    this.#solde = solde;
  }
  deposer(montant) {
    if (montant > 0) this.#solde += montant;
    return this;                         // permet d'enchaîner
  }
  retirer(montant) {
    if (montant > this.#solde) return false;
    this.#solde -= montant;
    return true;
  }
  get solde() { return this.#solde; }    // accesseur : s'utilise comme une propriété
  static creerVide(nom) { return new Compte(nom); }     // méthode de classe
}
const c = new Compte("Léa", 100);
c.deposer(50).deposer(10);
console.log(c.retirer(500), c.retirer(30), c.solde);
console.log(Compte.creerVide("Sam").solde);

2. L'héritage

class Animal {
  constructor(nom) { this.nom = nom; }
  crier() { return "..."; }
  presenter() { return `${this.nom} fait ${this.crier()}`; }
}
class Chien extends Animal {
  crier() { return "Ouaf"; }             // redéfinition
}
class Chiot extends Chien {
  constructor(nom, age) {
    super(nom);                          // appelle le constructeur de la mère
    this.age = age;
  }
  crier() { return super.crier() + " (aigu)"; }
}
for (const a of [new Animal("Truc"), new Chien("Rex"), new Chiot("Pif", 1)]) {
  console.log(a.presenter(), a instanceof Chien);
}

3. Le piège de this

this désigne l'objet sur lequel la méthode est appelée. Si on détache la méthode de son objet, this est perdu. Les fonctions fléchées, elles, gardent le this de l'endroit où elles sont écrites.

class Minuteur {
  constructor() { this.ticks = 0; }
  tick() { this.ticks++; }
  lancer(n) {
    for (let i = 0; i < n; i++) [1].forEach(() => this.tick());   // fléchée : this conservé
    return this.ticks;
  }
}
console.log(new Minuteur().lancer(3));

4. Les erreurs : try, catch, throw

function diviser(a, b) {
  if (b === 0) throw new Error("division par zéro");
  return a / b;
}
for (const b of [2, 0]) {
  try {
    console.log(diviser(10, b));
  } catch (e) {
    console.log("Erreur :", e.message);
  } finally {
    console.log("  (essai terminé)");
  }
}
try {
  JSON.parse("{pas du json}");
} catch (e) {
  console.log(e.name);                   // SyntaxError
}

5. Les modules

Un vrai projet est découpé en fichiers. Chaque fichier exporte ce qu'il veut partager, et importe ce dont il a besoin.

// fichier geometrie.js
export const PI = 3.14159;
export function aireCercle(r) { return PI * r * r; }
export default class Point { /* … */ }        // un export par défaut par fichier
// fichier main.js
import Point, { aireCercle, PI } from "./geometrie.js";
console.log(aireCercle(2));
<script type="module" src="main.js"></script>

Côté serveur, Node.js exécute JavaScript hors du navigateur ; npm installe les bibliothèques décrites dans package.json (npm install, npm run …).
