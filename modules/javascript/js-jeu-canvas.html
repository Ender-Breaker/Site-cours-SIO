1. La boucle de jeu du labo

let x = 50;

function setup() { x = 50; }             // une fois au démarrage (facultatif)

function update(dt) {                    // la logique, 60 fois par seconde
  if (touche("droite")) x += 200 * dt;   // 200 pixels par seconde
}

function draw() {                        // l'affichage
  effacer("black");
  rect(x, 140, 40, 40, "gold");
}
FonctionRôle
effacer(c)remplit l'écran (480 × 320)
rect(x, y, l, h, c), cercle(x, y, r, c), ligne(x1, y1, x2, y2, c)formes
texte(t, x, y, taille, c)texte
touche("gauche") / toucheAppuyee("espace")touche maintenue / appuyée à cette image
sourisX(), sourisY(), clic()souris
hasard(a, b), LARGEUR, HAUTEURentier au hasard, 480, 320

L'origine est en haut à gauche ; y augmente vers le bas.

2. Le vrai canvas du navigateur

Dans une page web, on dessine dans une balise <canvas> et on demande au navigateur de rappeler la fonction à chaque image :

const canvas = document.querySelector("canvas");
const ctx = canvas.getContext("2d");
const touches = {};
addEventListener("keydown", e => touches[e.key] = true);
addEventListener("keyup", e => touches[e.key] = false);

let x = 50, avant = performance.now();
function boucle(maintenant) {
  const dt = (maintenant - avant) / 1000;          // secondes écoulées
  avant = maintenant;
  if (touches["ArrowRight"]) x += 200 * dt;         // update
  ctx.fillStyle = "black";                          // draw
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "gold";
  ctx.fillRect(x, 140, 40, 40);
  requestAnimationFrame(boucle);                    // image suivante
}

let y = 260, vy = 0;
function update(dt) {
  if (toucheAppuyee("espace") && y >= 260) vy = -9;    // saut seulement au sol
  vy += 0.5;                                           // gravité
  y = Math.min(260, y + vy);
  if (y === 260) vy = 0;
}

4. Les états du jeu

let etat = "menu", score = 0;
function update(dt) {
  if (etat === "menu" && toucheAppuyee("espace")) { etat = "jeu"; score = 0; }
  else if (etat === "jeu") { score++; if (score > 300) etat = "fin"; }
  else if (etat === "fin" && toucheAppuyee("espace")) etat = "menu";
}
function draw() {
  effacer("black");
  if (etat === "menu") texte("ESPACE pour jouer", 130, 150, 24, "white");
  else if (etat === "jeu") texte("Score : " + score, 16, 16, 24, "white");
  else texte("Perdu ! Score : " + score, 130, 150, 24, "tomato");
}

Le vaisseau

const vaisseau = { x: 220, y: 280, l: 40, h: 20 };
function update(dt) {
  if (touche("gauche")) vaisseau.x -= 5;
  if (touche("droite")) vaisseau.x += 5;
  vaisseau.x = Math.max(0, Math.min(LARGEUR - vaisseau.l, vaisseau.x));
}
function draw() {
  effacer("#0d0e1a");
  rect(vaisseau.x, vaisseau.y, vaisseau.l, vaisseau.h, "deepskyblue");
}

Les météorites, la collision, le score

let meteores = [], chrono = 0, perdu = false, score = 0;

// dans update, avant le reste :
  if (perdu) return;
