1. Déclarer et appeler

function aire(largeur, hauteur) {
  return largeur * hauteur;
}
function saluer(nom = "inconnu") {       // valeur par défaut
  console.log(`Salut ${nom}`);
}
console.log(aire(3, 4));
saluer("Léa");
saluer();
console.log(aire(3));                    // NaN : hauteur vaut undefined

Une fonction sans return renvoie undefined. Un argument manquant vaut undefined, un argument en trop est ignoré.

2. Les fonctions fléchées

const carre = (x) => x * x;              // return implicite
const somme = (a, b) => {
  const s = a + b;
  return s;                              // avec des accolades, return explicite
};
const direBonjour = () => console.log("Bonjour");
console.log(carre(7), somme(3, 4));
direBonjour();

3. Les fonctions sont des valeurs

On peut passer une fonction en argument : c'est un callback. Tout JavaScript repose là-dessus (événements, tableaux, requêtes réseau).

function appliquer(f, valeur) {
  return f(valeur);
}
console.log(appliquer(x => x * 2, 21));
console.log(appliquer(Math.sqrt, 49));

function repeter(n, action) {
  for (let i = 1; i <= n; i++) action(i);
}
repeter(3, i => console.log(`tour ${i}`));

4. Portée et closures

Une variable let ou const n'existe que dans son bloc { }. Une fonction garde l'accès aux variables de l'endroit où elle a été créée : c'est une closure.

function creerCompteur() {
  let n = 0;                       // privée : inaccessible de l'extérieur
  return () => {
    n++;
    return n;
  };
}
const a = creerCompteur(), b = creerCompteur();
console.log(a(), a(), a(), b());   // 1 2 3 1 : chaque compteur a son propre n

5. Paramètres du reste et décomposition

function total(...nombres) {             // ...nombres : tous les arguments dans un tableau
  let s = 0;
  for (const n of nombres) s += n;
  return s;
}
console.log(total(1, 2, 3), total());

function minmax(t) { return [Math.min(...t), Math.max(...t)]; }   // ...t : étale le tableau
const [mini, maxi] = minmax([12, 5, 19]);                         // décomposition
console.log(mini, maxi);

6. Récursivité

function factorielle(n) {
  if (n <= 1) return 1;                  // cas d'arrêt
  return n * factorielle(n - 1);
}
console.log(factorielle(5), factorielle(10));

Une fonction qui reçoit un test

La même fonction sert pour tous les critères : ajoute le tien.

function compter(t, test) {
  let n = 0;
  for (const v of t) if (test(v)) n++;
  return n;
}
const notes = [12, 5, 19, 8, 10];
console.log(compter(notes, v => v >= 10));
console.log(compter(notes, v => v % 2 === 0));
